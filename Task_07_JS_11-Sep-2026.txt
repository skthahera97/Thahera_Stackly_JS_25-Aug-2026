Task1
Use this API:
https://fakestoreapi.com/products
Requirements:
Fetch all products.
Convert response using .json().
Display:
Product title
Price
Category
Use map() to create a new array containing only:
title
price
Use filter() to find products with price greater than $100.
Use find() to find the first product in the "electronics" category.
Use reduce() to calculate the total price.
Use sort() to arrange products from highest price to lowest.
Handle API errors using .catch().
Display a completion message using .finally().


const apiUrl = "https://fakestoreapi.com/products";

fetch(apiUrl)
  .then(res => res.json())
  .then(products => {

    // Display all products
    products.forEach(p =>
      console.log(`${p.title} | $${p.price} | ${p.category}`)
    );

    // map() - title & price
    const details = products.map(p => ({
      title: p.title,
      price: p.price
    }));
    console.log("Map:", details);

    // filter() - price > $100
    console.log("Filter:", products.filter(p => p.price > 100));

    // find() - first electronics product
    console.log("Find:", products.find(p => p.category === "electronics"));

    // reduce() - total price
    const total = products.reduce((sum, p) => sum + p.price, 0);
    console.log("Total:", `$${total.toFixed(2)}`);

    // sort() - highest to lowest
    const sorted = [...products].sort((a, b) => b.price - a.price);
    console.log("Sorted:", sorted);

  })
  .catch(error => console.log("API Error:", error))
  .finally(() => console.log("Completed"));

const apiUrl = "https://fakestoreapi.com/products";

const count = (products, category) =>
  products.filter(p => p.category === category).length;

fetch(apiUrl)
  .then(res => res.json())
  .then(products => {

    console.log("===== PRODUCT DASHBOARD =====");
    console.log("Total Products:", products.length);

    console.log("Electronics:", count(products, "electronics"));
    console.log("Jewelery:", count(products, "jewelery"));
    console.log("Men's Clothing:", count(products, "men's clothing"));
    console.log("Women's Clothing:", count(products, "women's clothing"));

    const prices = products.map(p => p.price);

    console.log("Highest Price:", `$${Math.max(...prices).toFixed(2)}`);
    console.log("Lowest Price:", `$${Math.min(...prices).toFixed(2)}`);

    const average = prices.reduce((sum, price) => sum + price, 0) / prices.length;
    console.log("Average Price:", `$${average.toFixed(2)}`);
  })
  .catch(error => console.log("Error:", error))
  .finally(() => console.log("Dashboard completed."));

Task 3 — User & Post API

Use these APIs:
https://jsonplaceholder.typicode.com/users
https://jsonplaceholder.typicode.com/posts
Requirements:
Fetch users.
Display all user names.
Display user name + email.
Find the user with ID 5.
Filter users from a particular city.
Fetch posts.
Display posts written by user ID 1.
Count how many posts user ID 1 has created.
Find the first post with more than 50 characters in the title.


const usersApi = "https://jsonplaceholder.typicode.com/users";
const postsApi = "https://jsonplaceholder.typicode.com/posts";

fetch(usersApi)
  .then(res => res.json())
  .then(users => {

    console.log("===== USERS =====");

    // Names
    users.forEach(u => console.log(u.name));

    // Name + Email
    users.forEach(u => console.log(`${u.name} - ${u.email}`));

    // User ID 5
    console.log("User 5:", users.find(u => u.id === 5));

    // Users from Gwenborough
    console.log(
      "Gwenborough:",
      users.filter(u => u.address.city === "Gwenborough")
    );

    return fetch(postsApi);
  })

  .then(res => res.json())
  .then(posts => {

    // Posts by User 1
    const user1Posts = posts.filter(p => p.userId === 1);

    user1Posts.forEach(p => console.log(p.title));

    console.log("User 1 Posts:", user1Posts.length);

    // First title > 50 characters
    console.log(
      "Long Title:",
      posts.find(p => p.title.length > 50)
    );
  })

  .catch(error => console.log("API Error:", error))
  .finally(() => console.log("Completed"));


  Task 4 — API + Search
Use:
https://fakestoreapi.com/products
Ask the user:
Enter product category:
Example:
electronics
Then display only products belonging to that category.
Also ask:
Enter maximum price:
Example:
500
Display products matching both conditions:
Category = electronics
Price <= 500
Concepts: prompt(), fetch(), filter(), conditions, functions.


const apiUrl = "https://fakestoreapi.com/products";

const category = prompt("Enter category:");
const maxPrice = Number(prompt("Enter maximum price:"));

fetch(apiUrl)
  .then(res => res.json())
  .then(products => {

    const results = products.filter(p =>
      p.category.toLowerCase() === category.toLowerCase() &&
      p.price <= maxPrice
    );

    console.log("===== SEARCH RESULTS =====");

    if (results.length === 0) {
      console.log("No products found.");
    } else {
      results.forEach(p =>
        console.log(`${p.title} - $${p.price}`)
      );
    }
  })
  .catch(error => console.log("API Error:", error))
  .finally(() => console.log("Search completed."));



  Task 5 — API Shopping Cart
Use:
https://fakestoreapi.com/products
Fetch the products and create a shopping cart.
The student should:
Display available products.
Select products using their IDs.
Add selected products to an array.
Calculate cart total using reduce().
Apply discount:
Above $100 → 10%
Above $200 → 20%
Display final amount.

const apiUrl = "https://fakestoreapi.com/products";

let cart = [];

fetch(apiUrl)
    .then(response => response.json())

    .then(products => {

        // 1. Display available products
        console.log("===== AVAILABLE PRODUCTS =====");

        products.forEach(product => {
            console.log(
                `ID: ${product.id} | ${product.title} | $${product.price}`
            );
        });

        // 2. Select product IDs
        const id1 = Number(prompt("Enter first product ID:"));
        const id2 = Number(prompt("Enter second product ID:"));

        // 3. Find and add selected products to cart
        const product1 = products.find(product => product.id === id1);
        const product2 = products.find(product => product.id === id2);

        if (product1) {
            cart.push(product1);
        }

        if (product2) {
            cart.push(product2);
        }

        // 4. Calculate total using reduce()
        const total = cart.reduce((sum, product) => {
            return sum + product.price;
        }, 0);

        // 5. Apply discount
        let discount = 0;

        if (total > 200) {
            discount = 20;
        } else if (total > 100) {
            discount = 10;
        }

        const discountAmount = total * discount / 100;
        const finalAmount = total - discountAmount;

        // 6. Display cart
        console.log("\n===== CART =====");

        cart.forEach((product, index) => {
            console.log(
                `Product ${index + 1}: ${product.title}`
            );
            console.log(
                `Price: $${product.price}`
            );
        });

        console.log(`\nTotal: $${total.toFixed(2)}`);
        console.log(`Discount: ${discount}%`);
        console.log(`Final Amount: $${finalAmount.toFixed(2)}`);
    })

    .catch(error => {
        console.log("Error:", error);
    });

const apiUrl = "https://fakestoreapi.com/products";

fetch(apiUrl)
    .then(response => response.json())
    .then(products => {

        console.log("===== PRODUCT REPORT =====");

        // Total products
        console.log("Total Products:", products.length);

        // map()
        const names = products.map(p => p.title);
        console.log("Product Names:", names);

        // filter()
        const expensive = products.filter(p => p.price > 100);
        console.log("Above $100:", expensive);

        // find()
        const electronics = products.find(
            p => p.category === "electronics"
        );
        console.log("Electronics:", electronics);

        // reduce()
        const total = products.reduce(
            (sum, p) => sum + p.price, 0
        );
        console.log("Total Value: $" + total.toFixed(2));

        // some()
        console.log(
            "Any Above $500:",
            products.some(p => p.price > 500)
        );

        // every()
        console.log(
            "All Above $1:",
            products.every(p => p.price > 1)
        );

        // sort()
        const sorted = [...products].sort(
            (a, b) => b.price - a.price
        );

        console.log("Highest to Lowest:");
        sorted.forEach(p =>
            console.log(`${p.title} - $${p.price}`)
        );

    })
    .catch(error => console.log("Error:", error))
    .finally(() => console.log("API request finished."));
