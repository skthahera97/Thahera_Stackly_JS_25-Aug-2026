    input: process.stdin,
    output: process.stdout
});

    num = Number(num);

    }

🟣 Generator Function
Task 31 — Cashback
Create a generator:
function* cashback() {
    yield "10% cashback";
    yield "20% cashback";
    yield "30% cashback";
    yield "Better luck next time";
}
Create the generator object and print all values using for...of.
This matches the generator approach in your syllabus.

function* cashback() {
    yield "10% cashback";
    yield "20% cashback";
    yield "30% cashback";
    yield "Better luck next time";
}

let result = cashback();

for (let value of result) {
    console.log(value);
}

Output :
10% cashback
20% cashback
30% cashback
Better luck next time
