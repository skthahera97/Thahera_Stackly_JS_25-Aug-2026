// Final Challenge — User + Calculator in Node.js

const readline = require("readline");

const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout
});

rl.question("Enter first number: ", function(num1) {

    rl.question("Enter second number: ", function(num2) {

        let a = Number(num1);
        let b = Number(num2);

        console.log("Addition:", a + b);
        console.log("Subtraction:", a - b);
        console.log("Multiplication:", a * b);
        console.log("Division:", a / b);
        console.log("Modulus:", a % b);
        console.log("Power:", a ** b);

        rl.close();
    });

});

// require("readline") → Load readline module | process.stdin → Get keyboard input 
// | process.stdout → Display console output 
// | rl.question() → Ask user for input | Number() → Convert string to number.
